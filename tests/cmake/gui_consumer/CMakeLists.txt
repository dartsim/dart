cmake_minimum_required(VERSION 3.22.1)
project(DARTGuiInstalledConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(DART CONFIG REQUIRED COMPONENTS dart)
if(NOT DART_INCLUDE_DIRS STREQUAL DART_PACKAGE_INCLUDE_DIRS)
  message(FATAL_ERROR "Core-only DART_INCLUDE_DIRS changed")
endif()
find_package(DART CONFIG REQUIRED COMPONENTS gui)
if(
  DART_TEST_USE_SYSTEM_IMGUI
  AND NOT DART_INCLUDE_DIRS STREQUAL DART_PACKAGE_INCLUDE_DIRS
)
  message(FATAL_ERROR "System-ImGui DART_INCLUDE_DIRS changed")
endif()
add_executable(gui_consumer main.cpp bare_imgui.cpp)
target_link_libraries(gui_consumer PRIVATE dart-gui)
enable_testing()
add_test(NAME gui_consumer COMMAND gui_consumer)
add_subdirectory(legacy)
find_package(DART CONFIG REQUIRED COMPONENTS dart)
if(NOT DART_INCLUDE_DIRS STREQUAL DART_PACKAGE_INCLUDE_DIRS)
  message(FATAL_ERROR "Core-only DART_INCLUDE_DIRS changed after finding GUI")
endif()
